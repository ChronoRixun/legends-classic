cmake_minimum_required(VERSION 3.20)
project(online_assets_lifetime LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The owned Alchemy runtime test needs MSVC Win32")
endif()
add_executable(online_assets_lifetime online_assets_lifetime.cpp)
target_compile_features(online_assets_lifetime PRIVATE cxx_std_17)
target_compile_options(online_assets_lifetime PRIVATE /UNDEBUG)
target_link_libraries(online_assets_lifetime PRIVATE psapi)
